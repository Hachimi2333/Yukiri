cmake_minimum_required(VERSION 3.28)
project(YukiriConsumer LANGUAGES CXX)
if(YUKIRI_SOURCE_DIR)
	set(YUKIRI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
	set(YUKIRI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
	add_subdirectory("${YUKIRI_SOURCE_DIR}" YukiriBuild EXCLUDE_FROM_ALL)
else()
	find_package(Yukiri 0.1 CONFIG REQUIRED)
endif()
add_executable(YukiriConsumer Main.cpp)
target_link_libraries(YukiriConsumer PRIVATE Yukiri::Yukiri)
